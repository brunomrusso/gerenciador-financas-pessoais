⚡ QUICK START (5 MINUTOS):

  # 1. Instalar backend
  python -m venv venv
  venv\Scripts\activate
  pip install -r requirements.txt
  copy .env.example .env
  # Editar .env com credenciais PostgreSQL
  python run.py

  # 2. Em outro terminal, instalar frontend
  cd frontend
  npm install
  npm run dev

  # 3. Abrir navegador
  http://localhost:5173
